**Goal:** Build the approved, clean-room `game.twentyfortyeight` MiniApp with deterministic 4×4 rules, retained MVIKotlin/Decompose state, versioned namespaced persistence, adaptive accessible Compose UI, and original procedural audio while keeping it discoverable and unshipped.

- [ ] **Step 3: Implement the value surface**

  Define these exact types:

  ```kotlin
  internal enum class Direction { Up, Down, Left, Right }

  internal data class Position(val row: Int, val column: Int) {
      init { require(row in 0..3 && column in 0..3) }
      val index: Int get() = row * 4 + column
  }

  @JvmInline
  internal value class TileValue(val value: Long) {
      init {
          require(value > 0L && (value and (value - 1L)) == 0L)
          require(value <= (1L shl 62))
      }
  }

  internal data class TileId(val value: Long)
  internal data class RuntimeTile(val id: TileId, val value: TileValue)
  internal data class RuntimeBoard(val tiles: List<RuntimeTile?>) {
      init { require(tiles.size == 16) }
      fun valueBoard(): Board = Board.fromValues(tiles.map { it?.value?.value })
      companion object {
          fun restore(board: Board): Pair<RuntimeBoard, Long>
      }
  }

  internal class Board private constructor(
      val values: List<Long?>,
  ) {
      companion object {
          const val Side = 4
          const val CellCount = 16
          fun empty(): Board
          fun fromValues(values: List<Long?>): Board
      }
      operator fun get(position: Position): TileValue?
      fun emptyPositions(): List<Position>
      fun maxTile(): TileValue?
      fun sum(): Long
  }
  ```

  `Board` defensively copies input and implements structural equality/hashCode; no mutable list escapes.

  internal data class RngState(
