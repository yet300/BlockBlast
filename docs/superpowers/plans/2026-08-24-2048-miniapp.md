# 2048 MiniApp Implementation Plan

> **Execution mode:** Inline Execution in this task, one checkbox at a time. Do not create a worktree or delegate to subagents. Stop at every stated gate or unexpected RED result rather than silently changing the design.

**Goal:** Build the approved, clean-room `game.twentyfortyeight` MiniApp with deterministic 4×4 rules, retained MVIKotlin/Decompose state, versioned namespaced persistence, adaptive accessible Compose UI, and original procedural audio while keeping it discoverable and unshipped.

**Architecture:** `:game:twentyfortyeight` owns pure engine/state, snapshots, the bounded persistence coordinator, Store, typed Labels, lifecycle-bound session adapter, internal navigation, session graph, UI, audio declarations, diagnostics, and tests. The generic `MiniAppSession.handleBack(): Boolean = false` hook lets Root offer toolbar/system/predictive Back to an active session before host exit without importing game types. App-scoped services are stateless; coordinator/Store/components/Label/audio adapters are `MiniAppSessionScope` objects, and only the adapter executes external side effects.

**Tech Stack:** Kotlin 2.4.10, Kotlin Multiplatform, Compose Multiplatform/Material 3, Decompose 3.5.0, MVIKotlin, Metro 1.4.1, kotlinx.coroutines/serialization, MiniAppStorage, MiniAppAudio, Compose Resources, kotlin-test, Compose UI test.

- [ ] **Step 3: Implement the value surface**

  Define these exact types:

  ```kotlin
  internal enum class Direction { Up, Down, Left, Right }

  internal data class Position(val row: Int, val column: Int) {
      init { require(row in 0..3 && column in 0..3) }
      val index: Int get() = row * 4 + column
  }

  @JvmInline
  internal value class TileValue(val value: Long) {
      init {
          require(value > 0L && (value and (value - 1L)) == 0L)
          require(value <= (1L shl 62))
      }
  }

  internal data class TileId(val value: Long)
  internal data class RuntimeTile(val id: TileId, val value: TileValue)
  internal data class RuntimeBoard(val tiles: List<RuntimeTile?>) {
      init { require(tiles.size == 16) }
      fun valueBoard(): Board = Board.fromValues(tiles.map { it?.value?.value })
      companion object {
          fun restore(board: Board): Pair<RuntimeBoard, Long>
      }
  }

  internal class Board private constructor(
      val values: List<Long?>,
  ) {
      companion object {
          const val Side = 4
          const val CellCount = 16
          fun empty(): Board
          fun fromValues(values: List<Long?>): Board
      }
      operator fun get(position: Position): TileValue?
      fun emptyPositions(): List<Position>
      fun maxTile(): TileValue?
      fun sum(): Long
  }
  ```

  `Board` defensively copies input and implements structural equality/hashCode; no mutable list escapes.

      val output = ArrayList<Long?>(Board.SIZE)
      var score = 0L
      var index = 0
      while (index < compact.size) {
          val current = compact[index]
          val next = compact.getOrNull(index + 1)
          if (next == current && current <= TileValue.MAX_MERGE_INPUT) {
              val merged = current * 2L // safe because MAX_MERGE_INPUT is checked above
              output += merged
              score = requireNotNull(checkedAdd(score, merged))
              index += 2
          } else {
              output += current
              index += 1
          }
      }
      while (output.size < Board.SIZE) output += null
      return LineReduction(output, sourceIds = emptyList(), scoreDelta = score)
  }
  ```

  ```kotlin
  fun apply(input: MoveInput, direction: Direction, transitionId: Long): MoveResult {
      if (direction !in GameRules.legalDirections(input.board)) {
          return MoveResult.Unchanged(direction, input.board, input.score, input.rng)
      }

      val mergeCount = countMerges(input.board, direction)
      // Reserve merge IDs, the spawn ID, and the next usable ID before reduction or RNG.
      checkedAdd(input.nextTileId, mergeCount.toLong() + 1L)
          ?: return MoveResult.Failed(direction, MoveFailure.IdentityOverflow)

      val oriented = orientedLines(input.board, direction) // four lists in travel order
      val reduced = oriented.map(::reduceRuntimeLine)
      val afterMove = rebuildRuntimeBoard(reduced, direction)

      val scoreDelta = checkedSum(reduced.flatMap { it.merges }.map { it.resultValue.value })
          ?: return MoveResult.Failed(direction, MoveFailure.ScoreOverflow)
      val scoreAfter = checkedAdd(input.score, scoreDelta)
          ?: return MoveResult.Failed(direction, MoveFailure.ScoreOverflow)
      val mergedBoard = afterMove.assignMergeIdsStartingAt(input.nextTileId)
      val spawnId = TileId(input.nextTileId + mergeCount)
      val spawned = spawnPolicy.spawn(mergedBoard.board, input.rng, spawnId)
      val finalBoard = mergedBoard.board.withTile(spawned.tile)
      return MoveResult.Changed(
          transitionId = transitionId,
          direction = direction,
          beforeBoard = input.board,
          afterMoveBoard = mergedBoard.board,
          finalBoard = finalBoard,
          motions = mergedBoard.motions,
          merges = mergedBoard.merges,
          scoreBefore = input.score,
          scoreDelta = scoreDelta,
          scoreAfter = scoreAfter,
          spawn = spawned.tile,
          rngBefore = input.rng,
          rngAfter = spawned.rng,

              VictoryTransition.FirstReached else VictoryTransition.None,
          gameOver = if (GameRules.legalDirections(finalBoard).isEmpty())

No list/channel/unbounded queue or `CoroutineScope` field exists. The MVIKotlin Executor launches `submit` in its existing `scope`; the coordinator itself neither stores nor creates a scope.

  private fun move(direction: Direction) {
      val current = state()
      if (current.bootstrap != BootstrapState.Ready || current.visibility != MiniAppVisibility.ACTIVE) return
      if (current.overlay != null) return
      if (current.activeTransition != null) {
          if (current.pendingDirection == null) dispatch(Msg.QueueDirection(direction))
          return
      }
      when (val result = engine.apply(current.requireMoveInput(), direction, nextTransitionId())) {
          is MoveResult.Unchanged -> Unit
          is MoveResult.Failed -> publish(
              TwentyFortyEightStore.Label.Diagnostic(
                  TwentyFortyEightFailure.InvariantViolation(
                      when (result.reason) {
                          MoveFailure.ScoreOverflow -> InvariantCode.ScoreOverflow
                          MoveFailure.IdentityOverflow -> InvariantCode.IdentityOverflow
                      },
                  ),
              ),
          )
          is MoveResult.Changed -> {
              val update = reduceChanged(current, result) // pure GameState/statistics/fact reservation
              dispatch(
                  Msg.MoveAccepted(
                      update.state,
                      VisualTransition.Move(result.transitionId, result),
                  ),
              )
              publishImmediateLabels(update.immediateLabels)
              submitCheckpoint(update.commit, update.reservedExternalLabels)
          }
      }
  }

  private fun undo() {
      val current = state()
      if (current.bootstrap != BootstrapState.Ready || current.visibility != MiniAppVisibility.ACTIVE) return
      if (current.overlay != null || current.activeTransition != null) return
      when (val result = GameRules.undo(current.requireRulesState())) {
          UndoResult.Unavailable -> Unit
          is UndoResult.Changed -> {
              val transition = VisualTransition.Undo(
                  transitionId = nextTransitionId(),
                  transition = result.transition,
              )
              dispatch(Msg.UndoAccepted(result.state, transition))
              submitCheckpoint(commitFor(result.state), reservedExternalLabels = emptyList())
          }
      }
  }

  private fun animationCompleted(id: Long) {
      val transition = state().activeTransition ?: return
      if (transition.transitionId != id) return
      val queued = state().pendingDirection
      dispatch(Msg.ClearTransitionAndQueue)
      if (queued != null) move(queued)
  }

      fun requestFocus(target: FocusTarget)

  internal class TwentyFortyEightSessionAdapter(
      private val navigation: SessionNavigation,
      private val audio: TwentyFortyEightAudioAdapter,
      private val analytics: TwentyFortyEightAnalytics,
      private val diagnostics: TwentyFortyEightDiagnostics,
      private val host: MiniAppSessionHost,

  internal fun resolveGesture(
      delta: Offset,
      velocity: Velocity,
      cancelled: Boolean,
      enabled: Boolean,
      startRegion: SwipeStartRegion,
      config: SwipeConfig,
  ): GestureDecision {
      if (cancelled || !enabled) return GestureDecision.Cancelled
      val crossedDistance = max(abs(delta.x), abs(delta.y)) >= config.distanceThresholdPx
      val crossedSlop = max(abs(delta.x), abs(delta.y)) >= config.touchSlopPx
      val crossedVelocity = max(abs(velocity.x), abs(velocity.y)) >= config.velocityThresholdPxPerSecond
      if (!crossedDistance && !(crossedSlop && crossedVelocity)) return GestureDecision.PendingTap
      val resolved = if (crossedDistance) delta else Offset(velocity.x, velocity.y)
      val horizontal = abs(resolved.x) >= abs(resolved.y)
      if (!horizontal && startRegion == SwipeStartRegion.VerticalScrollSupport) {
          return GestureDecision.DelegateVerticalScroll
      }

          if (resolved.x >= 0f) Direction.Right else Direction.Left
      } else {
          if (resolved.y >= 0f) Direction.Down else Direction.Up
      }
      return GestureDecision.GameMove(direction)
